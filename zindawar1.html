// لێرەدا پاسوۆرد هاتییە دیارکرن const CORRECT_PASSWORD = "JEKO78"; // فەرمان بۆ پشکنینا پاسوۆردێ function checkPassword() { const userInput = document.getElementById("password-input").value; const errorMsg = document.getElementById("error-message"); if (userInput === CORRECT_PASSWORD) { document.getElementById("password-section").classList.add("hidden"); document.getElementById("download-section").classList.remove("hidden"); } else { errorMsg.style.display = "block"; } } // فەرمانا ڕاستەقینە بۆ داونلۆدکرنا ڤیدیۆیا تیک تۆکێ ب بێ لۆگۆ async function startDownload() { const tiktokUrl = document.getElementById("tiktok-url").value; const downloadButton = document.querySelector("#download-section button"); if (tiktokUrl.trim() === "") { alert("هیڤییە بەستەرەکێ دروست یێ تیک تۆکێ بنووسە!"); return; } // گۆڕینی دەقێ دوگمێ بۆ دۆخی چاوەڕێکرنێ downloadButton.innerText = "...چاوەڕێبە"; downloadButton.disabled = true; try { // بەکارئینانا API یەکێ خێرا و بێبەرامبەر بۆ داونلۆدکرنێ const apiEndpoint = `https://www.tikwm.com/api/?url=${encodeURIComponent(tiktokUrl)}`; const response = await fetch(apiEndpoint); const result = await response.json(); if (result.code === 0 && result.data.play) { // وەرگرتنا بەستەرێ ڤیدیۆیێ ب بێ لۆگۆ const videoLink = result.data.play; // دروستکرنا لایەنەکێ کاتی بۆ داونلۆدکرنا ئۆتۆماتیکی const a = document.createElement("a"); a.href = videoLink; a.download = "tiktok_video.mp4"; document.body.appendChild(a); a.click(); document.body.removeChild(a); alert("پڕۆسە ب سەرکەفتینە ئەنجامدا! ڤیدیۆ دۆخی داونلۆدێ د دۆلپەڕێ تە دا گرت."); } else { alert("ببوورە! کێشەیەک د بەستەری دا هەیە یان ڤیدیۆ یا تایبەتە (Private)."); } } catch (error) { console.error("Error:", error); alert("کێشەیەک د هێڵێ دا هەیە، فەرموو دووبارە تاقی بکەوە."); } finally { // گەڕاندنەوا دوگمێ بۆ دۆخی ئاسایی downloadButton.innerText = "داونلۆدکرن"; downloadButton.disabled = false; } }
